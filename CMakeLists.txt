cmake_minimum_required(VERSION 3.16)
project(TaskForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(taskforge src/JobScheduler.cpp)
target_include_directories(taskforge PUBLIC include)
target_compile_options(taskforge PRIVATE -Wall -Wextra -Wpedantic)
add_executable(taskforge_demo src/main.cpp)
target_link_libraries(taskforge_demo PRIVATE taskforge)
add_executable(taskforge_tests tests/tests.cpp)
target_link_libraries(taskforge_tests PRIVATE taskforge)
enable_testing()
add_test(NAME taskforge_tests COMMAND taskforge_tests)
